<script>
(function() {
  const canvas = document.getElementById('gameCanvas');
  const ctx = canvas.getContext('2d');
  const gridSize = 20;
  const tileCount = canvas.width / gridSize;
  let snake, food, dx, dy, score, highScore, gameLoop, gameSpeed, gameOver;

  highScore = parseInt(localStorage.getItem('snakeHighScore') || '0');
  document.getElementById('highScore').textContent = highScore;

  function initGame() {
    snake = [{x: Math.floor(tileCount/2), y: Math.floor(tileCount/2)}];
    dx = 0; dy = 0;
    score = 0; gameSpeed = 120; gameOver = false;
    document.getElementById('score').textContent = score;
    spawnFood();
    drawGame();
  }

  function spawnFood() {
    do {
      food = {x: Math.floor(Math.random()*tileCount), y: Math.floor(Math.random()*tileCount)};
    } while(snake.some(s => s.x === food.x && s.y === food.y));
  }

  function drawGame() {
    // Background
    ctx.fillStyle = '#0d1117';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    // Grid (subtle)
    ctx.strokeStyle = '#161b22';
    ctx.lineWidth = 0.5;
    for(let i=0; i<tileCount; i++) {
      ctx.beginPath();
      ctx.moveTo(i*gridSize, 0); ctx.lineTo(i*gridSize, canvas.height);
      ctx.stroke();
      ctx.beginPath();
      ctx.moveTo(0, i*gridSize); ctx.lineTo(canvas.width, i*gridSize);
      ctx.stroke();
    }

    // Snake
    snake.forEach((seg, idx) => {
      const brightness = 1 - (idx / snake.length) * 0.5;
      if(idx === 0) {
        ctx.fillStyle = '#7ee787';
        ctx.shadowColor = '#7ee787';
        ctx.shadowBlur = 10;
      } else {
        ctx.fillStyle = `rgba(63, 185, 80, ${brightness})`;
        ctx.shadowBlur = 0;
      }
      ctx.fillRect(seg.x*gridSize+1, seg.y*gridSize+1, gridSize-2, gridSize-2);

      if(idx === 0) {
        // Eyes
        ctx.fillStyle = '#0d1117';
        ctx.shadowBlur = 0;
        let ex1, ey1, ex2, ey2;
        if(dx===1) { ex1=seg.x*gridSize+12; ey1=seg.y*gridSize+5; ex2=seg.x*gridSize+12; ey2=seg.y*gridSize+12; }
        else if(dx===-1) { ex1=seg.x*gridSize+5; ey1=seg.y*gridSize+5; ex2=seg.x*gridSize+5; ey2=seg.y*gridSize+12; }
        else if(dy===-1) { ex1=seg.x*gridSize+5; ey1=seg.y*gridSize+5; ex2=seg.x*gridSize+12; ey2=seg.y*gridSize+5; }
        else { ex1=seg.x*gridSize+5; ey1=seg.y*gridSize+12; ex2=seg.x*gridSize+12; ey2=seg.y*gridSize+12; }
        ctx.fillRect(ex1, ey1, 3, 3);
        ctx.fillRect(ex2, ey2, 3, 3);
      }
    });

    ctx.shadowBlur = 0;

    // Food with pulsing effect
    const pulse = Math.sin(Date.now()/200) * 2;
    ctx.fillStyle = '#f78166';
    ctx.shadowColor = '#f78166';
    ctx.shadowBlur = 8 + pulse;
    ctx.beginPath();
    ctx.arc(food.x*gridSize+gridSize/2, food.y*gridSize+gridSize/2, gridSize/2-2+pulse, 0, Math.PI*2);
    ctx.fill();
    ctx.shadowBlur = 0;

    // Game Over overlay
    if(gameOver) {
      ctx.fillStyle = 'rgba(13, 17, 23, 0.85)';
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.fillStyle = '#f85149';
      ctx.font = 'bold 28px monospace';
      ctx.textAlign = 'center';
      ctx.fillText('GAME OVER', canvas.width/2, canvas.height/2 - 15);
      ctx.fillStyle = '#8b949e';
      ctx.font = '16px monospace';
      ctx.fillText('Score: ' + score, canvas.width/2, canvas.height/2 + 15);
      ctx.fillStyle = '#58a6ff';
      ctx.font = '14px monospace';
      ctx.fillText('Click "Start / Restart"', canvas.width/2, canvas.height/2 + 40);
      ctx.textAlign = 'start';
    }
  }

  function update() {
    if(gameOver) return;

    const head = {x: snake[0].x + dx, y: snake[0].y + dy};

    // Wall collision
    if(head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount) {
      endGame(); return;
    }

    // Self collision
    if(snake.some(s => s.x === head.x && s.y === head.y)) {
      endGame(); return;
    }

    snake.unshift(head);

    // Eat food
    if(head.x === food.x && head.y === food.y) {
      score += 10;
      document.getElementById('score').textContent = score;
      if(score > highScore) {
        highScore = score;
        localStorage.setItem('snakeHighScore', highScore);
        document.getElementById('highScore').textContent = highScore;
      }
      spawnFood();
      // Speed up
      if(gameSpeed > 60) gameSpeed -= 2;
    } else {
      snake.pop();
    }

    drawGame();
  }

  function endGame() {
    gameOver = true;
    clearInterval(gameLoop);
    drawGame();
  }

  window.startGame = function() {
    if(gameLoop) clearInterval(gameLoop);
    initGame();
    gameLoop = setInterval(update, gameSpeed);
    canvas.focus();
  };

  document.addEventListener('keydown', function(e) {
    const key = e.key.toLowerCase();
    if((key==='arrowup'||key==='w') && dy!==1) { dx=0; dy=-1; e.preventDefault(); }
    if((key==='arrowdown'||key==='s') && dy!==-1) { dx=0; dy=1; e.preventDefault(); }
    if((key==='arrowleft'||key==='a') && dx!==1) { dx=-1; dy=0; e.preventDefault(); }
    if((key==='arrowright'||key==='d') && dx!==-1) { dx=1; dy=0; e.preventDefault(); }
  });

  // Touch controls for mobile
  let touchStartX, touchStartY;
  canvas.addEventListener('touchstart', function(e) {
    touchStartX = e.touches[0].clientX;
    touchStartY = e.touches[0].clientY;
    e.preventDefault();
  }, {passive:false});

  canvas.addEventListener('touchend', function(e) {
    if(!touchStartX || !touchStartY) return;
    const dx2 = e.changedTouches[0].clientX - touchStartX;
    const dy2 = e.changedTouches[0].clientY - touchStartY;
    if(Math.abs(dx2) > Math.abs(dy2)) {
      if(dx2 > 0 && dx!==-1) { dx=1; dy=0; }
      else if(dx2 < 0 && dx!==1) { dx=-1; dy=0; }
    } else {
      if(dy2 > 0 && dy!==-1) { dx=0; dy=1; }
      else if(dy2 < 0 && dy!==1) { dx=0; dy=-1; }
    }
    e.preventDefault();
  }, {passive:false});

  initGame();
})();
</script>
